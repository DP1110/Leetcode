1from collections import deque
2
3class Solution:
4    def weightedSum(self, parent, nums):
5        n = len(parent)
6        
7        # Build children adjacency list
8        children = [[] for _ in range(n)]
9        for i in range(1, n):
10            children[parent[i]].append(i)
11        
12        # BFS to compute depth of every node (root depth = 1)
13        depth = [0] * n
14        depth[0] = 1
15        queue = deque([0])
16        
17        while queue:
18            node = queue.popleft()
19            for child in children[node]:
20                depth[child] = depth[node] + 1
21                queue.append(child)
22        
23        h = max(depth)  # tree height
24        
25        # Calculate weighted sum
26        total = 0
27        for i in range(n):
28            total += nums[i] * (h - depth[i] + 1)
29        
30        return total
